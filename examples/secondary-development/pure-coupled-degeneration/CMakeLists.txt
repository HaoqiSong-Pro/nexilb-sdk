cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(nexilb_pure_coupled_degeneration_contract LANGUAGES C CXX)
endif()
if(NOT TARGET NexiLB::Headers)
  if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../../include/nexilb/nexilb.h")
    add_library(nexilb_example_headers INTERFACE)
    add_library(NexiLB::Headers ALIAS nexilb_example_headers)
    target_include_directories(nexilb_example_headers INTERFACE ../../../include)
  else()
    find_package(NexiLBHeaders 1.0.0 EXACT CONFIG REQUIRED)
  endif()
endif()
add_library(nexilb_pure_coupled_degeneration_contract OBJECT consumer_contract.cpp)
target_compile_features(nexilb_pure_coupled_degeneration_contract PRIVATE cxx_std_17)
target_include_directories(nexilb_pure_coupled_degeneration_contract PRIVATE ../include)
target_link_libraries(nexilb_pure_coupled_degeneration_contract PRIVATE NexiLB::Headers)
add_executable(nexilb_degeneration_consumer main.c)
target_compile_features(nexilb_degeneration_consumer PRIVATE c_std_11)
target_include_directories(nexilb_degeneration_consumer PRIVATE ../include)
target_link_libraries(nexilb_degeneration_consumer PRIVATE NexiLB::Headers ${CMAKE_DL_LIBS})
